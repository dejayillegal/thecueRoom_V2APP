8) GitHub Actions CI/Deploy (no Replit workflow)

Create/replace .github/workflows/ci.yml with the pipeline below (pnpm, Node 20, strict gates, and deploy):

name: CI

on:
  push: { branches: [ main, develop ] }
  pull_request: { branches: [ main, develop ] }
  workflow_dispatch:

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  NEXT_TELEMETRY_DISABLED: '1'
  NODE_ENV: production
  NEXT_PUBLIC_APP_NAME: thecueRoom
  NEXT_PUBLIC_ENV: ci

jobs:
  setup:
    runs-on: ubuntu-latest
    outputs:
      has_playwright: ${{ steps.detect.outputs.has_playwright }}
      has_web: ${{ steps.detect.outputs.has_web }}
      is_export: ${{ steps.detect.outputs.is_export }}
    steps:
      - uses: actions/checkout@v4
      - name: Enforce pnpm lockfile only
        run: |
          if [ -f package-lock.json ] || [ -f yarn.lock ]; then
            echo "Remove package-lock.json / yarn.lock. Use pnpm-lock.yaml only."
            exit 1
          fi
          if [ ! -f pnpm-lock.yaml ]; then echo "Generating pnpm-lock.yaml"; fi
      - uses: actions/setup-node@v4
        with: { node-version: 20.x, cache: 'pnpm' }
      - uses: pnpm/action-setup@v4
        with: { version: 9, run_install: false }
      - name: Install (create lockfile if missing)
        run: pnpm install --frozen-lockfile=false
      - name: Detect projects
        id: detect
        run: |
          has_web="false"; [ -d "apps/web" ] && has_web="true"
          has_pw="false"; ls -1 **/playwright.config.* >/dev/null 2>&1 && has_pw="true"
          is_export="false"; grep -R "output:\\s*'export'" apps/web/next.config.* >/dev/null 2>&1 && is_export="true"
          echo "has_web=$has_web" >> $GITHUB_OUTPUT
          echo "has_playwright=$has_pw" >> $GITHUB_OUTPUT
          echo "is_export=$is_export" >> $GITHUB_OUTPUT

  verify:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20.x, cache: 'pnpm' }
      - uses: pnpm/action-setup@v4
        with: { version: 9, run_install: false }
      - run: pnpm install --frozen-lockfile
      - name: Typecheck
        run: pnpm typecheck
      - name: Lint (no warnings)
        run: pnpm lint
      - name: Unit tests
        run: pnpm test
      - name: Dependency hygiene (strict)
        run: pnpm depcheck
      - name: Unused files (strict)
        run: pnpm unused

  e2e:
    if: needs.setup.outputs.has_playwright == 'true'
    runs-on: ubuntu-latest
    needs: [setup, verify]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20.x, cache: 'pnpm' }
      - uses: pnpm/action-setup@v4
        with: { version: 9, run_install: false }
      - run: pnpm install --frozen-lockfile
      - run: npx playwright install --with-deps
      - name: Build & start web (if present)
        run: |
          if [ -d "apps/web" ]; then
            pnpm --filter ./apps/web run build
            pnpm --filter ./apps/web run start &>/dev/null &
            sleep 3
          fi
      - name: Run Playwright
        run: npx playwright test --reporter=line

  build_web:
    if: needs.setup.outputs.has_web == 'true'
    runs-on: ubuntu-latest
    needs: [setup, verify]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20.x, cache: 'pnpm' }
      - uses: pnpm/action-setup@v4
        with: { version: 9, run_install: false }
      - run: pnpm install --frozen-lockfile
      - name: Build web
        working-directory: apps/web
        run: pnpm run build
      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: web-build
          path: |
            apps/web/.next
            apps/web/out
            apps/web/reports
          if-no-files-found: warn

  deploy_web_export:
    name: Deploy Web (GitHub Pages, export)
    if: needs.setup.outputs.has_web == 'true' && needs.setup.outputs.is_export == 'true'
    runs-on: ubuntu-latest
    needs: [build_web]
    permissions: { contents: write, pages: write, id-token: write }
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20.x, cache: 'pnpm' }
      - uses: pnpm/action-setup@v4
        with: { version: 9, run_install: false }
      - run: pnpm install --frozen-lockfile
      - name: Export static
        working-directory: apps/web
        run: pnpm run build && pnpm exec next export
      - name: Deploy to gh-pages
        uses: peaceiris/actions-gh-pages@v4
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          publish_dir: ./apps/web/out
          force_orphan: true

  deploy_web_vercel:
    name: Deploy Web (Vercel, dynamic)
    if: needs.setup.outputs.has_web == 'true' && needs.setup.outputs.is_export != 'true'
    runs-on: ubuntu-latest
    needs: [build_web]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20.x }
      - name: Vercel Deploy
        run: |
          npm i -g vercel@latest
          vercel pull --yes --environment=production --token=${{ secrets.VERCEL_TOKEN }}
          vercel build --prod --token=${{ secrets.VERCEL_TOKEN }}
          vercel deploy --prebuilt --prod --token=${{ secrets.VERCEL_TOKEN }}
        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}
          VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID }}
          VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID }}
