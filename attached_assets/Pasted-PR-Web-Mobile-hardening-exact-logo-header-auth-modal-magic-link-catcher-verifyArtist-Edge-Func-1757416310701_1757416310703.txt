  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
);

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [status, setStatus] = useState<"idle"|"pending"|"approved"|"rejected"|"needs-review">("idle");
  useEffect(() => { if (open) catchHashAuth(supabase).catch(()=>{}); }, [open]);

  async function onSubmit(e: React.FormEvent) {
    e.preventDefault();
    const { data, error } = await supabase.auth.signInWithPassword({ email, password });

    // Call verifyArtist Edge Function (client stub below)

apps/web/lib/hashAuth.ts
import type { SupabaseClient } from "@supabase/supabase-js";

/**
 * Catches Supabase magic link tokens from URL hash (#access_token=...)
 * and completes the session without navigating away. Safe on static hosts.
 */
export async function catchHashAuth(supabase: SupabaseClient) {
  if (typeof window === "undefined") return;
  const hash = window.location.hash;
  if (!hash || !hash.includes("access_token")) return;

  const params = new URLSearchParams(hash.replace(/^#/, ""));
  const access_token = params.get("access_token");
  const refresh_token = params.get("refresh_token");
  if (!access_token) return;

  await supabase.auth.setSession({ access_token, refresh_token: refresh_token ?? "" });
  // Clean the URL so the hash isn’t preserved
  history.replaceState(null, "", window.location.pathname + window.location.search);
}

// Deno Edge Function (Supabase): verifyArtist
// Heuristics: domain allow-list, social/profile presence, follower thresholds.
// Writes into public.profiles (approval_status, approval_reason).
import "jsr:@supabase/functions-js/edge-runtime.d.ts";

type Input = { user_id?: string; profiles_row?: { website?: string|null; instagram?: string|null; soundcloud?: string|null; followers?: number|null } };
type Verdict = "approved" | "needs-review" | "rejected";

const ALLOW_DOMAINS = ["soundcloud.com","bandcamp.com","mixcloud.com","instagram.com","x.com","twitter.com","youtube.com","linktr.ee"];
const MIN_FOLLOWERS = 200; // tune as needed

Deno.serve(async (req) => {
  const url = new URL(req.url);
  if (req.method !== "POST") return new Response("Method Not Allowed", { status: 405 });

  const supabaseUrl = Deno.env.get("SUPABASE_URL")!;
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
  const auth = req.headers.get("authorization") ?? "";
  if (!auth.startsWith("Bearer ")) return new Response("Unauthorized", { status: 401 });

  const { user_id, profiles_row }: Input = await req.json().catch(() => ({} as any));

  const { createClient } = await import("npm:@supabase/supabase-js@2");
  const admin = createClient(supabaseUrl, serviceKey);

  // load profile if not provided
  const profile = profiles_row ?? (await admin
      .from("profiles")
      .select("id,user_id,website,instagram,soundcloud,followers")
      .eq("user_id", user_id)
      .maybeSingle()).data ?? {};

  // heuristics
  const urls = [profile.website, profile.instagram, profile.soundcloud].filter(Boolean) as string[];
  const hasAllowed = urls.some(u => {
    try { return ALLOW_DOMAINS.includes(new URL(u).hostname.replace(/^www\./,"")); }
    catch { return false; }
  });

  const followerScore = typeof profile.followers === "number" ? profile.followers : 0;
  let verdict: Verdict = "needs-review";
  let reason = "Queued for curator review";

  if (hasAllowed && followerScore >= MIN_FOLLOWERS) {
    verdict = "approved";
    reason = "Auto-approved by heuristics";
  } else if (!hasAllowed && followerScore < 20) {
    verdict = "rejected";
    reason = "No recognized artist presence";
  }

  // persist
  await admin.from("profiles")
    .update({ approval_status: verdict, approval_reason: reason, approval_checked_at: new Date().toISOString() })
    .eq("user_id", user_id!);

  return Response.json({ approval_status: verdict, approval_reason: reason });
});

export async function requestVerifyArtist(userId: string, accessToken: string) {
  const res = await fetch(`${process.env.NEXT_PUBLIC_SUPABASE_URL}/functions/v1/verifyArtist`, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${accessToken}`,
      "Content-Type": "application/json",
      apikey: process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,

docs/AUTH_FLOW.md (short)
# Auth flow (modal + hash catcher)

1. User clicks **Login / Sign Up** → `AuthModal` opens (no route change).
2. Email+password sign in uses Supabase.
3. If using “Email Link”, Supabase sends magic link; the **hash fragment** with tokens is caught by `catchHashAuth()` (no server required on GH Pages).
4. After sign-in, client POSTs to `verifyArtist` Edge Function with `Authorization: Bearer <access_token>`.
5. Heuristics return `approved | needs-review | rejected`; we show `ApprovalBanner`. Only `approved` users get the gated UI.
